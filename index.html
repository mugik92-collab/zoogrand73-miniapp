from aiogram import Bot, Dispatcher, types from aiogram.filters import CommandStart import asyncio TOKEN="ТОКЕН_ОТ_BOTFATHER" bot=Bot(TOKEN) dp=Dispatcher() @dp.message(CommandStart()) async def start(msg:types.Message): kb=types.ReplyKeyboardMarkup( keyboard=[[types.KeyboardButton( text="🐾 Открыть ZooGrand", web_app=types.WebAppInfo(url="ССЫЛКА_НА_INDEX.HTML") )]], resize_keyboard=True ) await msg.answer("Добро пожаловать в ZooGrand!",reply_markup=kb) @dp.message() async def data(msg:types.Message): if msg.web_app_data: await msg.answer(f"📩 Новая заявка:\n{msg.web_app_data.data}") async def main(): await dp.start_polling(bot) asyncio.run(main())
